cmake_minimum_required(VERSION 3.24)
project(vulkan_sandbox VERSION 0.1.0 LANGUAGES C CXX)

option(SANDBOX_BUILD_SAMPLES "Build the Windows/Android reference application" ON)
option(SANDBOX_ENABLE_IMGUI "Build optional ImGui sample UI" ON)
option(SANDBOX_BUILD_TESTS "Build CPU-only regression tests" ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/vma/include/vk_mem_alloc.h")
    message(FATAL_ERROR "Missing VMA. Run git submodule update --init --recursive")
endif()
if(ANDROID)
    # Vulkan comes from the NDK target, never the host Windows Vulkan SDK.
    find_library(SANDBOX_ANDROID_VULKAN vulkan REQUIRED)
    add_library(Vulkan::Vulkan UNKNOWN IMPORTED GLOBAL)
    set_target_properties(Vulkan::Vulkan PROPERTIES IMPORTED_LOCATION "${SANDBOX_ANDROID_VULKAN}")
else()
    find_package(Vulkan 1.3 REQUIRED)
endif()
add_subdirectory(engine)

if(SANDBOX_BUILD_SAMPLES)
    set(sample_dependencies glm/CMakeLists.txt)
    if(SANDBOX_ENABLE_IMGUI)
        list(APPEND sample_dependencies imgui/imgui.h)
    endif()
    foreach(dependency IN LISTS sample_dependencies)
        if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${dependency}")
            message(FATAL_ERROR "Missing ${dependency}. Run git submodule update --init --recursive")
        endif()
    endforeach()
    include(SandboxShaders)
    set(GLM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLM_BUILD_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory(third_party/glm EXCLUDE_FROM_ALL)
    add_subdirectory(samples/viewer)
    if(ANDROID)
        add_subdirectory(platforms/android)
    elseif(WIN32)
        add_subdirectory(platforms/windows)
    else()
        message(FATAL_ERROR "Sample adapters currently support Windows and Android. Use SANDBOX_BUILD_SAMPLES=OFF to consume the core elsewhere.")
    endif()
endif()

if(SANDBOX_BUILD_TESTS AND NOT ANDROID)
    enable_testing()
    add_subdirectory(tests)
endif()
